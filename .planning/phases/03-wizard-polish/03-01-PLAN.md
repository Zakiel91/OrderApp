```typescript
import { useEffect, useState } from 'react'
import { useLanguage } from '../context/LanguageContext'

interface Props {
  form: unknown
}

export function DraftSavedToast({ form }: Props) {
  const { t } = useLanguage()
  const [visible, setVisible] = useState(false)

  useEffect(() => {
    const debounce = setTimeout(() => {
      setVisible(true)
      const hide = setTimeout(() => setVisible(false), 2000)
      return () => clearTimeout(hide)
    }, 500)
    return () => clearTimeout(debounce)
  }, [form])

  if (!visible) return null

  return (
    <div
      className="fixed top-4 left-1/2 -translate-x-1/2 z-40
                 bg-[var(--color-success)] text-white text-[14px]
                 px-4 py-1.5 rounded-full
                 animate-[fadeIn_150ms_ease-out]"
    >
      &#x2713; {t('draft_saved')}
    </div>
  )
}
```

Key implementation notes (per D-03 through D-08):
- Component receives `form` as a prop (type `unknown` is intentional — caller passes the form object, component only uses it as a dependency trigger for useEffect)
- `useEffect` dependency is `[form]` — fires whenever form object reference changes (i.e., any field update)
- Debounce: 500ms after last change before showing; 2000ms visible duration
- Positioning: `fixed top-4 left-1/2 -translate-x-1/2 z-40` — RTL-safe centering, below BottomNav's z-50
- Background: `bg-[var(--color-success)]` — never hardcoded hex
- Checkmark: `&#x2713;` — Unicode U+2713, not an emoji (per UI-SPEC constraint 4)
- No shadow, no border, no close button (per D-03 and UI-SPEC)
- `animate-[fadeIn_150ms_ease-out]` requires the `fadeIn` keyframe in index.css (added in Task 2)
- Exit is handled by `setVisible(false)` returning null — fade-out animation is acceptable as instant for now (no exit animation needed to satisfy requirements)
