**handleFieldSave:**
```typescript
const handleFieldSave = async (field: string, value: string) => {
  if (!order) return
  setSaving(true)
  try {
    const payload: Record<string, unknown> = { id: order.id, [field]: value }
    if (field === 'price_to_client' || field === 'advance_amount') {
      payload[field] = value ? parseFloat(value) : undefined
    }
    await updateOrder(payload)
    setOrder(prev => prev ? { ...prev, [field]: field === 'price_to_client' || field === 'advance_amount' ? (value ? parseFloat(value) : undefined) : value } : prev)
    setSheet(null)
    setSaveSuccess(true)
    setTimeout(() => setSaveSuccess(false), 2000)
  } finally {
    setSaving(false)
  }
}
```

        {editable && (
